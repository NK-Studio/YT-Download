var
  DesktopShortcutCheck: TNewCheckBox;

// 폴더 선택 화면에 체크박스 UI 덧붙이기
procedure InitializeWizard;
begin
  // 체크박스 객체 생성
  DesktopShortcutCheck := TNewCheckBox.Create(WizardForm);
  
  // 폴더 선택 페이지(SelectDirPage)를 부모로 지정하여 해당 화면에 종속시킴
  DesktopShortcutCheck.Parent := WizardForm.SelectDirPage;
  
  // 체크박스 위치 및 크기 조정 (폴더 경로 입력칸 바로 아래로 자연스럽게 배치)
  DesktopShortcutCheck.Left := WizardForm.DirEdit.Left;
  DesktopShortcutCheck.Top := WizardForm.DirEdit.Top + WizardForm.DirEdit.Height + 24;
  DesktopShortcutCheck.Width := 300;
  
  // 체크박스 텍스트 및 기본 상태 설정
  DesktopShortcutCheck.Caption := '바탕화면에 바로가기 만들기';
  DesktopShortcutCheck.Checked := True; // 기본으로 체크된 상태
end;

// [Icons] 섹션에서 바탕화면 아이콘을 만들지 말지 결정하는 함수
function ShouldCreateDesktopIcon: Boolean;
begin
  // 사용자가 체크박스를 체크했는지 여부를 반환
  Result := DesktopShortcutCheck.Checked;
end;
